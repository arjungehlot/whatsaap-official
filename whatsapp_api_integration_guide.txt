const axios = require("axios"); // or import axios from "axios";

const WHATSAPP_API_URL = "https://whatsaapapi.nearmeteacher.in/api/v1/message/create";
const WHATSAPP_API_KEY = process.env.WHATSAPP_API_KEY || "wf_live_h7exzBR8KFIBPzr78W4xqN9yuaISEYCWd3rnvZGy2Pw";

/**
 * Core function to send WhatsApp message via NearMeTeacher API
 */
async function sendWhatsAppMessage({ to, message, fileUrl = null, buttons = null }) {
  try {
    const payload = {
      receiverMobileNo: String(to).replace(/\D/g, "").slice(-10),
    };

    if (buttons) {
      payload.button = buttons;
    } else {
      payload.message = Array.isArray(message) ? message : [message];
    }

    if (fileUrl) {
      payload.filePathUrl = Array.isArray(fileUrl) ? fileUrl : [fileUrl];
    }

    const response = await axios.post(WHATSAPP_API_URL, payload, {
      headers: {
        "Content-Type": "application/json",
        "x-api-key": WHATSAPP_API_KEY,
      },
      timeout: 15000,
    });

    return response.data;
  } catch (error) {
    console.error("WhatsApp Send Error:", error.response?.data || error.message);
    return null;
  }
}

const { notifyTeacherOnEnquiry } = require("../services/whatsapp.service");

async function handleStudentEnquiry(req, res) {
  // 1. Save enquiry to database
  const enquiry = await Enquiry.create(req.body);
  const teacher = await Teacher.findById(enquiry.teacherId);

  // 2. Trigger WhatsApp notification (asynchronous, non-blocking)
  notifyTeacherOnEnquiry({
    teacherPhone: teacher.phone,
    teacherName: teacher.name,
    studentName: enquiry.studentName,
    subject: enquiry.subject,
    gradeOrClass: enquiry.grade,
    location: enquiry.location || "Online",
  }).catch((err) => console.error("WhatsApp error:", err));

  return res.status(201).json({ success: true, enquiry });
}

const { notifyNotesUploaded } = require("../services/whatsapp.service");

async function handleNotesUpload(req, res) {
  const note = await Note.create({
    title: req.body.title,
    subject: req.body.subject,
    fileUrl: req.file.location, // S3 or server URL
  });

  const students = await Student.find({ subject: note.subject });
  for (const student of students) {
    notifyNotesUploaded({
      studentPhone: student.phone,
      studentName: student.name,
      teacherName: req.user.name,
      subject: note.subject,
      notesTitle: note.title,
      notesPdfUrl: note.fileUrl,
    });
  }

  return res.json({ success: true, note });
}
